### 디스코드 봇 개발 중지 및 텔레그램 봇 개발로 전향
디스코드 봇을 개발하면서 원한 것은 지정된 시간마다 디스코드 알림이 오는 것이었는데, 디스코드 봇은 그러한 기능에 적합하지 않다고 판단하게 되었다. 또한, 우리학교 공지사항은 li로 안 묶여 있고.. table 형태로 묶여있어서 그런지 디스코드 알림에 엔터가 엄청나게 많이 쳐져서 왔다... 그것을 보고 이것은 안 되겠다 싶어 마침 찾아보던 중 텔레그램에도 마찬가지로 봇 기능이 있다고 하여 텔레그램 봇으로 개발을 다시 진행할 예정이다. 마찬가지로 PyCharm으로 진행 예정이다.

### 중지된 개발 코드
~~~python
import discord
from discord.ext import commands
import requests
from bs4 import BeautifulSoup  # 뷰티풀스프와 리퀘스트 임포트 해오기 이때 Beautifulsoup4를 다운받아야함
import datetime

app = commands.Bot(command_prefix='/', intents=discord.Intents.all())  # 명령어는 /로 시작
today = datetime.date.today()

@app.event
async def on_ready():
    print('Done')
    await app.change_presence(status=discord.Status.online, activity=None)


@app.command()
async def test(ctx):  
    try:
        testresponse = requests.get("테스트사이트") #테스트사이트 부분에 원하는 사이트를 입력하면 됨
        # testresponse라는 이름으로 사이트 get 해오기

        testresponse.encoding = 'utf-8'
        testhtml = testresponse.text
        testsoup = BeautifulSoup(testhtml, 'html.parser')
        # 모듈을 가지고 오고 requests 모듈을 사용하여 해당 URL에서 HTML 소스를 가지고 옴. 가지고 온 소스의 인코딩 변환을 해주고, BeautifulSoup 모듈로 데이터를 파싱함.

        test정보 = testsoup.select('tr')  # test 정보라는 이름으로 tr요소들을 전부

        for 정보 in test정보[1:10]:
            분류, 번호, 제목, 작성자, 작성일, 조회 = 정보.select('td')
            await ctx.send(
                f'```기관 : test 공지\n제목 : {제목.text}\n날짜 : {작성일.text}```\n링크 : 사이트{제목.a["href"]}\n--------------------------------------')
    except:
        await ctx.send(f'{today.month}월 {today.day}일 {today.hour}시 {today.minute}분 test 오류 발생')


app.run('본인의 토큰')
~~~ 
여기까지 코드를 작성하였고, 아래 사이트를 보고 자신이 원하는 사이트와 이에 맞춰 필요한 것을 가져오면 된다고 생각한다. 본인의 대학교의 공지사항은 해당 형태와 맞지 않는다면 이에 맞춰 수정해야한다.
